\begin{theorem*}[Image of a Preimage]
Let $f:A\to B$ and let $T\subseteq B$. Then
\[
f(f^{-1}(T)) = T\cap \operatorname{im}(f).
\]
In particular, if $T\subseteq \operatorname{im}(f)$, then
\[
f(f^{-1}(T))=T.
\]
\end{theorem*}
