\subsection{Generated Collections and Closure Systems}

\begin{exposition}
Many later structures are defined by starting with a small family of sets and
closing it under specified operations.  The topology generated by a basis, the
$\sigma$-algebra generated by a collection, and the Borel $\sigma$-algebra
generated by open sets are all instances of the same pattern:
\[
\text{generated object}
=
\bigcap\{\text{closed objects containing the generators}\}.
\]
The point is not cardinality.  The point is control: generation gives the
smallest structure containing the data one starts with.
\end{exposition}

\begin{definitionbox}{Definition (Collection System)}
\begin{definition}[Collection System]\label{def:collection-system}
Let $X$ be a set. A \emph{collection system} on $X$ is a family whose members
are collections of subsets of $X$:
\[
\mathfrak{S}\subseteq \mathcal{P}(\mathcal{P}(X)).
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\mathfrak S\text{ is a collection system on }X
\iff
\mathfrak S\subseteq\mathcal P(\mathcal P(X)).
\]
\end{remark*}

\begin{remark*}[Interpretation]
A member of $\mathfrak{S}$ is itself a collection of subsets of $X$.  This is
the level at which topologies, set algebras, and $\sigma$-algebras live.
\end{remark*}

\begin{definitionbox}{Definition (Closure System)}
\begin{definition}[Closure System]\label{def:closure-system}
A collection system $\mathfrak{S}\subseteq\mathcal{P}(\mathcal{P}(X))$ is a
\emph{closure system} if arbitrary intersections of members of $\mathfrak{S}$
again belong to $\mathfrak{S}$.
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\forall \mathcal{T}\subseteq\mathfrak{S},\qquad
\bigcap_{\mathcal{A}\in\mathcal{T}}\mathcal{A}\in\mathfrak{S}.
\]
\end{remark*}

\begin{remark*}[Interpretation]
Closure systems are exactly the families for which the "intersection of all
eligible structures" construction stays inside the family.
\end{remark*}

\begin{definitionbox}{Definition (Generated Collection)}
\begin{definition}[Generated Collection]\label{def:generated-collection}
Let $\mathfrak{S}$ be a closure system on $X$, and let
$\mathcal{E}\subseteq\mathcal{P}(X)$ be a family of generators.  The
\emph{collection generated by $\mathcal{E}$ in $\mathfrak{S}$} is
\[
\langle\mathcal{E}\rangle_{\mathfrak{S}}
=
\bigcap\{\mathcal{A}\in\mathfrak{S}:\mathcal{E}\subseteq\mathcal{A}\}.
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\langle\mathcal E\rangle_{\mathfrak S}
=
\bigcap\{\mathcal A\in\mathfrak S:\mathcal E\subseteq\mathcal A\}.
\]
\end{remark*}

\begin{remark*}[Interpretation]
The generated collection is the smallest $\mathfrak{S}$-closed collection that
contains the generators.
\end{remark*}

\begin{theorembox}{Theorem (Generated Collection Closure Laws)}
\begin{theorem}[Generated Collection Closure Laws]\label{thm:generated-collection-closure-laws}
Let $\mathfrak{S}$ be a closure system and let
$\mathcal{E}\subseteq\mathcal{P}(X)$.  Suppose there is at least one
$\mathcal{A}\in\mathfrak{S}$ with $\mathcal{E}\subseteq\mathcal{A}$.  Then:
\begin{enumerate}[label=(\roman*)]
\item $\mathcal{E}\subseteq\langle\mathcal{E}\rangle_{\mathfrak{S}}$;
\item if $\mathcal{E}\subseteq\mathcal{F}$, then
      $\langle\mathcal{E}\rangle_{\mathfrak{S}}
      \subseteq
      \langle\mathcal{F}\rangle_{\mathfrak{S}}$;
\item $\langle\mathcal{E}\rangle_{\mathfrak{S}}\in\mathfrak{S}$;
\item
      $\langle\langle\mathcal{E}\rangle_{\mathfrak{S}}\rangle_{\mathfrak{S}}
      =
      \langle\mathcal{E}\rangle_{\mathfrak{S}}$.
\end{enumerate}
\hyperref[prf:generated-collection-closure-laws]{Go to proof.}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\begin{aligned}
\mathcal E&\subseteq\langle\mathcal E\rangle_{\mathfrak S},\\
\mathcal E\subseteq\mathcal F&\to
\langle\mathcal E\rangle_{\mathfrak S}\subseteq
\langle\mathcal F\rangle_{\mathfrak S},\\
\langle\mathcal E\rangle_{\mathfrak S}&\in\mathfrak S,\\
\langle\langle\mathcal E\rangle_{\mathfrak S}\rangle_{\mathfrak S}
&=
\langle\mathcal E\rangle_{\mathfrak S}.
\end{aligned}
\]
\end{remark*}

\begin{remark*}[Interpretation]
Generation acts like a closure operator: it is extensive, monotone, lands in
the closed objects, and is idempotent.
\end{remark*}
