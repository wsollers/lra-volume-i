\begin{lraaxiomlist}
\LraAxiomItem{\hyperref[ax:extensionality]{Extensionality}}

  {the constant \(\varnothing\) has no members}
\LraAxiomItem{\hyperref[ax:pairing]{Pairing}}
  {given \(a,b\), the set \(\{a,b\}\) exists}
\LraAxiomItem{\hyperref[ax:union]{Union}}

\LraAxiomItem{\hyperref[ax:power-set]{Power Set}}

\LraAxiomItem{\hyperref[ax:infinity]{Infinity}}

\LraAxiomItem{\hyperref[ax:separation]{Separation Schema}}

\LraAxiomItem{\hyperref[ax:replacement]{Replacement Schema}}

\LraAxiomItem{\hyperref[ax:choice]{Choice}}
