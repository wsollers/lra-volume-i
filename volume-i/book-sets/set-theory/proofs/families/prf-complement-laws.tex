\begin{theorem*}[Complement Extreme and Complementarity Laws]
Let $A\subseteq U$. Then
\[
\varnothing^c=U,
\qquad
U^c=\varnothing,
\]
\[
A\cup A^c=U,
\qquad
A^c\cup A=U,
\]
and
\[
A\cap A^c=\varnothing,
\qquad
A^c\cap A=\varnothing.
\]
\end{theorem*}
