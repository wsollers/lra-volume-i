\begin{theorem*}[Identity and Absorption Laws]
Let $A,B$ be sets and $U$ a universe with $A \subseteq U$. Then
\[
A \cup \varnothing = A,
\qquad
A \cap U = A,
\]
\[
A \cup (A \cap B) = A,
\qquad
A \cap (A \cup B) = A.
\]
\end{theorem*}
